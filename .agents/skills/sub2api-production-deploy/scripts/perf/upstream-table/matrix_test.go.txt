type counters struct {
	queries, rows                                        atomic.Int64
	adminQueries, adminRows, gatewayQueries, gatewayRows atomic.Int64
}
type routeCounterKey struct{}

func (c *counters) query(ctx context.Context) string {
	c.queries.Add(1)
	route, _ := ctx.Value(routeCounterKey{}).(string)
	if route == "admin" {
		c.adminQueries.Add(1)
	} else if route == "gateway" {
		c.gatewayQueries.Add(1)
	}
	return route
}

type budgetLogCounter struct{ count atomic.Int64 }

func (c *budgetLogCounter) Write(p []byte) (int, error) {
	if strings.Contains(string(p), "upstream_cache_guard_budget_exhausted") {
		c.count.Add(1)
	}
	return len(p), nil
}

type connector struct {
	driver.Connector
	c *counters
}

func (c connector) Connect(ctx context.Context) (driver.Conn, error) {
	v, e := c.Connector.Connect(ctx)
	if e != nil {
		return nil, e
	}
	return conn{v, c.c}, nil
}

type conn struct {
	driver.Conn
	c *counters
}

func (c conn) IsValid() bool {
	if v, ok := c.Conn.(driver.Validator); ok {
		return v.IsValid()
	}
	return true
}
func (c conn) ResetSession(ctx context.Context) error {
	if v, ok := c.Conn.(driver.SessionResetter); ok {
		return v.ResetSession(ctx)
	}
	return nil
}
func (c conn) BeginTx(ctx context.Context, o driver.TxOptions) (driver.Tx, error) {
	c.c.query(ctx)
	return c.Conn.(driver.ConnBeginTx).BeginTx(ctx, o)
}
func (c conn) ExecContext(ctx context.Context, q string, a []driver.NamedValue) (driver.Result, error) {
	c.c.query(ctx)
	return c.Conn.(driver.ExecerContext).ExecContext(ctx, q, a)
}
func (c conn) QueryContext(ctx context.Context, q string, a []driver.NamedValue) (driver.Rows, error) {
	route := c.c.query(ctx)
	r, e := c.Conn.(driver.QueryerContext).QueryContext(ctx, q, a)
	if e != nil {
		return nil, e
	}
	return rows{r, c.c, route}, nil
}

type rows struct {
	driver.Rows
	c     *counters
	route string
}

func (r rows) Next(dest []driver.Value) error {
	e := r.Rows.Next(dest)
	if e == nil {
		r.c.rows.Add(1)
		if r.route == "admin" {
			r.c.adminRows.Add(1)
		} else if r.route == "gateway" {
			r.c.gatewayRows.Add(1)
		}
	}
	return e
}

func envInt(k string, fallback int) int {
	n, e := strconv.Atoi(os.Getenv(k))
	if e != nil {
		return fallback
	}
	return n
}
func must(t *testing.T, e error) {
	t.Helper()
	if e != nil {
		t.Fatal(e)
	}
}
func execSQL(t *testing.T, db *sql.DB, q string, args ...any) {
	t.Helper()
	_, e := db.Exec(q, args...)
	must(t, e)
}

const matrixSyntheticSSE = `data: {"id":"matrix","object":"chat.completion.chunk","model":"gpt-test","choices":[{"index":0,"delta":{"role":"assistant","content":"ok"},"finish_reason":null}]}

data: {"id":"matrix","object":"chat.completion.chunk","model":"gpt-test","choices":[{"index":0,"delta":{},"finish_reason":"stop"}],"usage":{"prompt_tokens":100,"completion_tokens":1,"total_tokens":101,"prompt_tokens_details":{"cached_tokens":90}}}

data: [DONE]

`

// A committed HTTP 200 is not sufficient evidence of a completed stream.
// Validate the fixed synthetic upstream's client-visible termination and usage;
// errors contain only classifications, never response payloads.
func validateMatrixSSECompletion(body []byte) error {
	done, finished := false, false
	usageCount := 0
	for _, raw := range bytes.Split(body, []byte("\n")) {
		line := bytes.TrimSpace(raw)
		if bytes.Equal(line, []byte("event: error")) {
			return fmt.Errorf("stream error event")
		}
		if !bytes.HasPrefix(line, []byte("data:")) {
			continue // Empty lines and the handler's optional comment heartbeats.
		}
		if done {
			return fmt.Errorf("stream data after completion")
		}
		data := bytes.TrimSpace(bytes.TrimPrefix(line, []byte("data:")))
		if bytes.Equal(data, []byte("[DONE]")) {
			done = true
			continue
		}
		var chunk struct {
			Error   json.RawMessage `json:"error"`
			Choices []struct {
				FinishReason *string `json:"finish_reason"`
			} `json:"choices"`
			Usage *struct {
				PromptTokens        *int `json:"prompt_tokens"`
				CompletionTokens    *int `json:"completion_tokens"`
				TotalTokens         *int `json:"total_tokens"`
				PromptTokensDetails *struct {
					CachedTokens *int `json:"cached_tokens"`
				} `json:"prompt_tokens_details"`
			} `json:"usage"`
		}
		if err := json.Unmarshal(data, &chunk); err != nil {
			return fmt.Errorf("invalid stream chunk")
		}
		if len(chunk.Error) > 0 && !bytes.Equal(bytes.TrimSpace(chunk.Error), []byte("null")) {
			return fmt.Errorf("stream error payload")
		}
		for _, choice := range chunk.Choices {
			if choice.FinishReason != nil && *choice.FinishReason == "stop" {
				finished = true
			}
		}
		if usage := chunk.Usage; usage != nil {
			usageCount++
			if usage.PromptTokens == nil || *usage.PromptTokens != 100 ||
				usage.CompletionTokens == nil || *usage.CompletionTokens != 1 ||
				usage.TotalTokens == nil || *usage.TotalTokens != 101 ||
				usage.PromptTokensDetails == nil || usage.PromptTokensDetails.CachedTokens == nil ||
				*usage.PromptTokensDetails.CachedTokens != 90 {
				return fmt.Errorf("unexpected synthetic stream usage")
			}
		}
	}
	if !done || !finished || usageCount != 1 {
		return fmt.Errorf("incomplete synthetic stream")
	}
	return nil
}

func readMatrixResponseBody(reader io.Reader, discard bool) ([]byte, error) {
	if discard {
		_, err := io.Copy(io.Discard, reader)
		return nil, err
	}
	return io.ReadAll(reader)
}

type matrixReadError struct{}

func (matrixReadError) Read([]byte) (int, error) { return 0, io.ErrUnexpectedEOF }

func TestMatrixResponseValidation(t *testing.T) {
	for _, test := range []struct {
		name  string
		body  string
		valid bool
	}{
		{"complete", matrixSyntheticSSE, true},
		{"comment heartbeats and CRLF", ":\r\n\r\n" + strings.ReplaceAll(matrixSyntheticSSE, "\n", "\r\n"), true},
		{"missing done", strings.ReplaceAll(matrixSyntheticSSE, "data: [DONE]", ""), false},
		{"missing finish", strings.ReplaceAll(matrixSyntheticSSE, `"finish_reason":"stop"`, `"finish_reason":null`), false},
		{"missing usage", "data: {\"choices\":[{\"finish_reason\":\"stop\"}]}\n\ndata: [DONE]\n\n", false},
		{"wrong input", strings.ReplaceAll(matrixSyntheticSSE, `"prompt_tokens":100`, `"prompt_tokens":99`), false},
		{"wrong output", strings.ReplaceAll(matrixSyntheticSSE, `"completion_tokens":1`, `"completion_tokens":2`), false},
		{"wrong total", strings.ReplaceAll(matrixSyntheticSSE, `"total_tokens":101`, `"total_tokens":102`), false},
		{"wrong cache", strings.ReplaceAll(matrixSyntheticSSE, `"cached_tokens":90`, `"cached_tokens":0`), false},
		{"missing cache", strings.ReplaceAll(matrixSyntheticSSE, `"cached_tokens":90`, `"cached_tokens":null`), false},
		{"malformed chunk", "data: {\n\ndata: [DONE]\n\n", false},
		{"error payload", "data: {\"error\":{\"type\":\"upstream_error\"}}\n\n" + matrixSyntheticSSE, false},
		{"error event", "event: error\n" + matrixSyntheticSSE, false},
		{"data after done", matrixSyntheticSSE + "data: {}\n\n", false},
		{"duplicate done", matrixSyntheticSSE + "data: [DONE]\n\n", false},
		{"duplicate usage", strings.ReplaceAll(matrixSyntheticSSE, "data: [DONE]", matrixSyntheticSSE), false},
	} {
		t.Run(test.name, func(t *testing.T) {
			if err := validateMatrixSSECompletion([]byte(test.body)); (err == nil) != test.valid {
				t.Fatalf("stream validation accepted=%v want=%v", err == nil, test.valid)
			}
		})
	}
	for _, discard := range []bool{false, true} {
		reader := io.MultiReader(strings.NewReader(matrixSyntheticSSE), matrixReadError{})
		if _, err := readMatrixResponseBody(reader, discard); err == nil {
			t.Fatalf("body read error ignored: discard=%v", discard)
		}
		if _, err := readMatrixResponseBody(strings.NewReader(matrixSyntheticSSE), discard); err != nil {
			t.Fatalf("valid body read failed: discard=%v", discard)
		}
	}
}

	var lastReadinessSample time.Time
	for time.Since(readinessStart) < time.Duration(envInt("MATRIX_READINESS_SECONDS", 20))*time.Second {
		initialView = service.GetUpstreamCacheGuardAccountView(seededAccounts[0])
		initialActive = initialView.Summary != nil && initialView.Summary.RatePercent != nil && service.PerfMatrixCacheSnapshot(1, 1).Fresh
		if lastReadinessSample.IsZero() || time.Since(lastReadinessSample) >= time.Second {
			stats := service.PerfMatrixCacheStats()
			stats["elapsed_ms"] = time.Since(readinessStart).Milliseconds()
			readinessTimeline = append(readinessTimeline, stats)
			encodedProgress, _ := json.Marshal(stats)
			fmt.Printf("MATRIX_READINESS_POPULATION=%s\n", encodedProgress)
			initialAllReady = stats["fresh"] == int64(n)
			lastReadinessSample = time.Now()
		}
		if initialActive && initialAllReady {
			break
		}
		time.Sleep(200 * time.Millisecond)
	}
	readinessSeconds := time.Since(readinessStart).Seconds()
	if initializationCPU != nil {
		pprof.StopCPUProfile()
		initializationCPU.Close()
	}
	if !initialActive && historical <= 1000 {
		t.Fatalf("cache protection inactive after readiness: %+v", initialView.Summary)
	}
	fmt.Printf("MATRIX_INITIAL_CACHE=%+v active=%v ready_seconds=%.3f\n", initialView.Summary, initialActive, readinessSeconds)
	var seq atomic.Int64
	var measuring atomic.Bool
	var listTimesMu sync.Mutex
	listTimes := []time.Duration{}
	request := func(admin bool) (time.Duration, int, string) {
		start := time.Now()
		var req *http.Request
		if admin {
			params := url.Values{"scope": {"upstream"}, "lite": {"1"}, "page": {strconv.Itoa(envInt("MATRIX_PAGE_NUMBER", 1))}, "page_size": {strconv.Itoa(page)}}
			if os.Getenv("MATRIX_VERSION") == "optimized" {
				params.Set("include_quality", "1")
			}
			if filter := os.Getenv("MATRIX_QUALITY_FILTER"); filter != "" {
				params.Set("quality_filter", filter)
			}
			req, _ = http.NewRequest("GET", app.URL+"/admin/upstream/accounts?"+params.Encode(), nil)
		} else {
			id := seq.Add(1) % 32
			body := fmt.Sprintf(`{"model":"gpt-test","messages":[{"role":"user","content":"hello"}],"stream":true,"prompt_cache_key":"matrix-session-%d","stream_options":{"include_usage":true}}`, id)
			req, _ = http.NewRequest("POST", app.URL+"/v1/chat/completions", bytes.NewBufferString(body))
			req.Header.Set("Content-Type", "application/json")
		}
		resp, err := httpClient.Do(req)
		if err != nil {
			return time.Since(start), 0, err.Error()
		}
		b, readErr := readMatrixResponseBody(resp.Body, false)
		resp.Body.Close()
		if readErr != nil {
			return time.Since(start), 0, "response body read failed"
		}
		if !admin && resp.StatusCode == http.StatusOK {
			if err := validateMatrixSSECompletion(b); err != nil {
				return time.Since(start), 0, err.Error()
			}
		}
		if admin && measuring.Load() {
			listTimesMu.Lock()
			listTimes = append(listTimes, time.Since(start))
			listTimesMu.Unlock()
		}
		if admin && resp.StatusCode == 200 {
			var payload struct {
				Data struct {
					Items []struct {
						ID int64 `json:"id"`
					} `json:"items"`
					QualityStatus string `json:"quality_stats_status"`
				} `json:"data"`
			}
			if err := json.Unmarshal(b, &payload); err != nil {
				return time.Since(start), 0, "invalid list JSON"
			}
			if os.Getenv("MATRIX_VERSION") == "optimized" && payload.Data.QualityStatus != "ok" {
				return time.Since(start), 0, "quality unavailable"
			}
			if os.Getenv("MATRIX_VERSION") == "baseline" {
				ids := make([]int64, 0, len(payload.Data.Items))
				for _, item := range payload.Data.Items {
					ids = append(ids, item.ID)
				}
				body, _ := json.Marshal(map[string]any{"account_ids": ids})
				qreq, _ := http.NewRequest("POST", app.URL+"/admin/accounts/quality-stats/batch", bytes.NewReader(body))
				qreq.Header.Set("Content-Type", "application/json")
				qresp, qerr := httpClient.Do(qreq)
				if qerr != nil {
					return time.Since(start), 0, qerr.Error()
				}
				_, readErr := readMatrixResponseBody(qresp.Body, true)
				qresp.Body.Close()
				if readErr != nil {
					return time.Since(start), 0, "quality body read failed"
				}
				if qresp.StatusCode != 200 {
					return time.Since(start), qresp.StatusCode, "quality failed"
				}
			}
		}
		return time.Since(start), resp.StatusCode, string(b)
	}
	for _, adminRead := range []bool{true, false} {
		_, status, body := request(adminRead)
		if status != 200 {
			t.Fatalf("warmup admin=%v status=%d body=%s", adminRead, status, body)
		}
	}
	if scenario == "slow" {
		execSQL(t, db, `CREATE OR REPLACE FUNCTION matrix_slow_runtime() RETURNS trigger LANGUAGE plpgsql AS $$ BEGIN PERFORM pg_sleep(0.04); RETURN NEW; END $$; CREATE TRIGGER matrix_slow_runtime BEFORE INSERT OR UPDATE ON upstream_cache_guard_runtime FOR EACH ROW EXECUTE FUNCTION matrix_slow_runtime()`)
	}
	if scenario == "full" {
		pipe := rdb.Pipeline()
		for id := int64(1); id <= int64(n); id++ {
			members := make([]redis.Z, 100)
			for i := range members {
				members[i] = redis.Z{Score: float64(time.Now().Add(time.Hour).Unix()), Member: fmt.Sprintf("matrix-owned-full-%d", i)}
			}
			pipe.ZAdd(context.Background(), fmt.Sprintf("concurrency:upstream:%d", id), members...)
		}
		_, err := pipe.Exec(context.Background())
		must(t, err)
		acquired, err := conc.AcquireTargetSlot(context.Background(), seededAccounts[0].SchedulingConcurrencyTarget())
		must(t, err)
		if acquired.Acquired {
			acquired.ReleaseFunc()
			t.Fatal("full scenario failed to exhaust actual target capacity")
		}
	}
	var completedBefore int
	must(t, db.QueryRow(`SELECT count(*) FROM upstream_cache_guard_facts WHERE processed=true AND evidence_id NOT LIKE 'cleanup-%'`).Scan(&completedBefore))
	upstreamBefore := upstreamRequests.Load()
	budgetBefore := budgetLogs.count.Load()
	var before, after runtime.MemStats
	runtime.GC()
	runtime.ReadMemStats(&before)
	// PostgreSQL table access statistics are database-wide for this exclusive
	// fixture; these count tuple visits rather than result rows or exact plan scans.
	pgStats := func() [2]int64 {
		var stats [2]int64
		must(t, db.QueryRow(`SELECT coalesce(sum(seq_tup_read),0)::bigint,coalesce(sum(idx_tup_fetch),0)::bigint FROM pg_stat_all_tables WHERE schemaname='public'`).Scan(&stats[0], &stats[1]))
		return stats
	}
	pgBefore := pgStats()
	pgIO := func() [4]int64 {
		var value [4]int64
		must(t, db.QueryRow(`SELECT temp_files,temp_bytes,deadlocks,blks_read FROM pg_stat_database WHERE datname=current_database()`).Scan(&value[0], &value[1], &value[2], &value[3]))
		return value
	}
	pgIOBefore := pgIO()
	sqlBefore, rowsBefore := counts.queries.Load(), counts.rows.Load()
	adminSQLBefore, adminRowsBefore := counts.adminQueries.Load(), counts.adminRows.Load()
	gatewaySQLBefore, gatewayRowsBefore := counts.gatewayQueries.Load(), counts.gatewayRows.Load()
	runtime.SetMutexProfileFraction(10)
	runtime.SetBlockProfileRate(1000000)
	if outdir != "" {
		must(t, os.MkdirAll(outdir, 0700))
		f, err := os.Create(filepath.Join(outdir, "cpu.pprof"))
		must(t, err)
		must(t, pprof.StartCPUProfile(f))
		defer f.Close()
		defer pprof.StopCPUProfile()
	}
	start := time.Now()
	var cpuBefore, cpuAfter syscall.Rusage
	must(t, syscall.Getrusage(syscall.RUSAGE_SELF, &cpuBefore))
	if outdir != "" {
		must(t, os.WriteFile(filepath.Join(outdir, "measurement.start"), []byte(start.UTC().Format(time.RFC3339Nano)), 0600))
	}
	deadline := start.Add(time.Duration(seconds) * time.Second)
	measuring.Store(true)
	var wg sync.WaitGroup
	var lock sync.Mutex
	gatewayTimes, adminTimes := []time.Duration{}, []time.Duration{}
	gatewayErrors, adminErrors := 0, 0
	refreshSamples, refreshUnavailable, refreshStale := 0, 0, 0
	refreshPopulationTimeline := []map[string]int64{}
	var maxRefreshLagMS int64
	var pgLockWaitSamples, pgLockWaitMaximum, pgWaitReadErrors int64
	wg.Add(1)
	go func() {
		defer wg.Done()
		for time.Now().Before(deadline) {
			var waiting int64
			waitCtx, cancelWait := context.WithTimeout(context.Background(), time.Second)
			waitErr := db.QueryRowContext(waitCtx, `SELECT count(*) FROM pg_stat_activity WHERE datname=current_database() AND pid<>pg_backend_pid() AND wait_event_type='Lock'`).Scan(&waiting)
			cancelWait()
			snapshot := service.PerfMatrixCacheSnapshot(1, 1)
			population := service.PerfMatrixCacheStats()
			population["elapsed_ms"] = time.Since(start).Milliseconds()
			lock.Lock()
			if waitErr != nil {
				pgWaitReadErrors++
			} else {
				pgLockWaitSamples += waiting
				if waiting > pgLockWaitMaximum {
					pgLockWaitMaximum = waiting
				}
			}
			refreshPopulationTimeline = append(refreshPopulationTimeline, population)
			refreshSamples++
			if !snapshot.Available {
				refreshUnavailable++
			} else if !snapshot.Fresh {
				refreshStale++
			}
			if !snapshot.LoadedAt.IsZero() {
				lag := time.Since(snapshot.LoadedAt).Milliseconds()
				if lag > maxRefreshLagMS {
					maxRefreshLagMS = lag
				}
			}
			lock.Unlock()
			pause := time.Until(deadline)
			if pause > time.Second {
				pause = time.Second
			}
			if pause > 0 {
				time.Sleep(pause)
			}
		}
	}()
	gatewayStatuses := map[int]int{}
	fullExpectedRejections, fullUnexpectedResponses, fullInvalidJSON := 0, 0, 0
	fullSignatures := map[string]int{}
	fullMismatches := []map[string]any{}
	for worker := 0; worker < envInt("MATRIX_GATEWAY_WORKERS", 8); worker++ {
		wg.Add(1)
		go func() {
			defer wg.Done()
			for time.Now().Before(deadline) {
				elapsed, status, body := request(false)
				lock.Lock()
				gatewayTimes = append(gatewayTimes, elapsed)
				gatewayStatuses[status]++
				if scenario == "full" {
					var payload struct {
						Error *struct {
							Type    string `json:"type"`
							Code    string `json:"code"`
							Message string `json:"message"`
						} `json:"error"`
					}
					parseErr := json.Unmarshal([]byte(body), &payload)
					typeName, code, message := "", "", ""
					if payload.Error != nil {
						typeName, code, message = payload.Error.Type, payload.Error.Code, payload.Error.Message
					}
					expected := parseErr == nil && status == http.StatusTooManyRequests && typeName == "rate_limit_error" && code == "gateway_queue_full" && message == "Too many pending requests, please retry later"
					if expected {
						fullExpectedRejections++
					} else {
						fullUnexpectedResponses++
						if parseErr != nil {
							fullInvalidJSON++
						}
						if len(fullMismatches) < 5 {
							fullMismatches = append(fullMismatches, map[string]any{"status": status, "type": typeName, "code": code, "invalid_json": parseErr != nil})
						}
					}
					signature := fmt.Sprintf("%d/%s/%s", status, typeName, code)
					if len(fullSignatures) < 16 || fullSignatures[signature] > 0 {
						fullSignatures[signature]++
					} else {
						fullSignatures["other"]++
					}
				}
				if status != 200 {
					gatewayErrors++
				}
				lock.Unlock()
			}
		}()
	}
	for worker := 0; worker < envInt("MATRIX_ADMINS", 20); worker++ {
		wg.Add(1)
		go func() {
			defer wg.Done()
			next := start
			for time.Now().Before(deadline) {
				if delay := time.Until(next); delay > 0 {
					time.Sleep(delay)
				}
				if !time.Now().Before(deadline) {
					return
				}
				elapsed, status, _ := request(true)
				lock.Lock()
				adminTimes = append(adminTimes, elapsed)
				if status != 200 {
					adminErrors++
				}
				lock.Unlock()
				next = next.Add(5 * time.Second)
				// One request per administrator at a time; missed slots never
				// accumulate a burst of catch-up refreshes.
				for !next.After(time.Now()) {
					next = next.Add(5 * time.Second)
				}
			}
		}()
	}
	wg.Wait()
	must(t, syscall.Getrusage(syscall.RUSAGE_SELF, &cpuAfter))
	if outdir != "" {
		must(t, os.WriteFile(filepath.Join(outdir, "measurement.end"), []byte(time.Now().UTC().Format(time.RFC3339Nano)), 0600))
	}
	runtime.ReadMemStats(&after)
	measurementSeconds := time.Since(start).Seconds()
	measuredCalls, measuredRows := counts.queries.Load()-sqlBefore, counts.rows.Load()-rowsBefore
	time.Sleep(1100 * time.Millisecond)
	pgAfter := pgStats()
	pgIOAfter := pgIO()
	percentiles := func(values []time.Duration) map[string]any {
		sort.Slice(values, func(i, j int) bool { return values[i] < values[j] })
		out := map[string]any{"samples": len(values)}
		out["p99_eligible"] = len(values) >= 1000
		for _, p := range []int{50, 95, 99} {
			if len(values) > 0 && (p != 99 || len(values) >= 1000) {
				out[fmt.Sprintf("p%d_ms", p)] = float64(values[(len(values)-1)*p/100]) / float64(time.Millisecond)
			}
		}
		return out
	}
	var pending int
	var oldest sql.NullTime
	must(t, db.QueryRow(`SELECT count(*),min(completed_at) FROM upstream_cache_guard_facts WHERE processed=false`).Scan(&pending, &oldest))
	var completed int
	must(t, db.QueryRow(`SELECT count(*) FROM upstream_cache_guard_facts WHERE processed=true`).Scan(&completed))
	var cleanupRemaining int
	must(t, db.QueryRow(`SELECT count(*) FROM upstream_cache_guard_facts WHERE evidence_id LIKE 'cleanup-%'`).Scan(&cleanupRemaining))
	accounts, e := accountRepo.GetByIDs(context.Background(), []int64{1})
	must(t, e)
	view := service.GetUpstreamCacheGuardAccountView(accounts[0])
	finalSnapshot := service.PerfMatrixCacheSnapshot(1, 1)
	result := map[string]any{"accounts": n, "history": historical, "page": page, "scenario": scenario, "wall_seconds": measurementSeconds, "gateway": percentiles(gatewayTimes), "admin": percentiles(adminTimes), "gateway_errors": gatewayErrors, "admin_errors": adminErrors, "sql_driver_calls": measuredCalls, "sql_returned_rows": measuredRows, "allocated_bytes": after.TotalAlloc - before.TotalAlloc, "allocations": after.Mallocs - before.Mallocs, "heap_alloc_bytes": after.HeapAlloc, "pending_completion": pending, "completed_facts": completed, "account1_cache": view.Summary, "gc_count": after.NumGC - before.NumGC,
		"measurement_process_cpu_seconds": float64(cpuAfter.Utime.Sec-cpuBefore.Utime.Sec+cpuAfter.Stime.Sec-cpuBefore.Stime.Sec) + float64(cpuAfter.Utime.Usec-cpuBefore.Utime.Usec+cpuAfter.Stime.Usec-cpuBefore.Stime.Usec)/1e6,
		"pg_temp_files":                   pgIOAfter[0] - pgIOBefore[0], "pg_temp_bytes": pgIOAfter[1] - pgIOBefore[1], "pg_deadlocks": pgIOAfter[2] - pgIOBefore[2], "pg_blocks_read": pgIOAfter[3] - pgIOBefore[3],
		"pg_lock_wait_connection_samples": pgLockWaitSamples, "pg_max_waiting_connections": pgLockWaitMaximum, "pg_wait_read_errors": pgWaitReadErrors,
		"list_http": percentiles(listTimes), "admin_refresh_period_seconds": 5, "admin_workers": envInt("MATRIX_ADMINS", 20),
		"admin_sql_driver_calls": counts.adminQueries.Load() - adminSQLBefore, "admin_sql_returned_rows": counts.adminRows.Load() - adminRowsBefore,
		"gateway_sql_driver_calls": counts.gatewayQueries.Load() - gatewaySQLBefore, "gateway_sql_returned_rows": counts.gatewayRows.Load() - gatewayRowsBefore,
		"initial_cache_active": initialActive, "initial_cache": initialView.Summary, "readiness_seconds": readinessSeconds,
		"initial_all_accounts_fresh": initialAllReady, "readiness_population_timeline": readinessTimeline, "refresh_population_timeline": refreshPopulationTimeline, "final_population": service.PerfMatrixCacheStats(),
		"refresh_health_samples": refreshSamples, "refresh_unavailable_samples": refreshUnavailable, "refresh_stale_samples": refreshStale, "max_observed_account1_refresh_lag_ms": maxRefreshLagMS, "final_snapshot_available": finalSnapshot.Available, "final_snapshot_fresh": finalSnapshot.Fresh,
		"gateway_statuses": gatewayStatuses, "upstream_requests": upstreamRequests.Load() - upstreamBefore,
		"full_expected_rejections": fullExpectedRejections, "full_unexpected_responses": fullUnexpectedResponses, "full_invalid_json": fullInvalidJSON, "full_response_signatures": fullSignatures, "full_first_mismatches": fullMismatches,
		"generated_completed_facts": completed - completedBefore - cleanupRemaining,
		"budget_fallback_logs":      budgetLogs.count.Load() - budgetBefore,
		"cleanup_seeded":            1200, "cleanup_remaining": cleanupRemaining, "pg_seq_tuple_visits": pgAfter[0] - pgBefore[0], "pg_index_tuple_fetches": pgAfter[1] - pgBefore[1]}
	if !finalSnapshot.LoadedAt.IsZero() {
		result["final_account1_refresh_lag_ms"] = time.Since(finalSnapshot.LoadedAt).Milliseconds()
	}
	if view.Summary != nil && view.Summary.UpdatedAt != nil {
		result["account1_business_evidence_age_ms"] = time.Since(*view.Summary.UpdatedAt).Milliseconds()
	}
	var pgIdentityUpdated sql.NullTime
	must(t, db.QueryRow(`SELECT max(updated_at) FROM upstream_cache_guard_accounts WHERE account_id=1`).Scan(&pgIdentityUpdated))
	if pgIdentityUpdated.Valid {
		result["account1_persisted_state_age_ms"] = time.Since(pgIdentityUpdated.Time).Milliseconds()
	}
	if oldest.Valid {
		result["oldest_pending_age_ms"] = time.Since(oldest.Time).Milliseconds()
	}
	if outdir != "" {
		// Keep samples so pooled P99 eligibility is based on actual observations.
		durationsMS := func(values []time.Duration) []float64 {
			out := make([]float64, len(values))
			for i, v := range values {
				out[i] = float64(v) / float64(time.Millisecond)
			}
			return out
		}
		raw, err := json.Marshal(map[string][]float64{"admin_ms": durationsMS(adminTimes), "list_ms": durationsMS(listTimes), "gateway_ms": durationsMS(gatewayTimes)})
		must(t, err)
		must(t, os.WriteFile(filepath.Join(outdir, "latency_samples.json"), raw, 0600))
		for _, name := range []string{"mutex", "block", "allocs"} {
			f, err := os.Create(filepath.Join(outdir, name+".pprof"))
			must(t, err)
			must(t, pprof.Lookup(name).WriteTo(f, 0))
			f.Close()
		}
	}
	encoded, _ := json.Marshal(result)
	fmt.Printf("MATRIX_RESULT=%s\n", encoded)
	if scenario == "full" && upstreamRequests.Load() != upstreamBefore {
		t.Error("full target sent real upstream request")
	}
	if scenario == "full" && (len(gatewayTimes) == 0 || fullExpectedRejections != len(gatewayTimes) || fullUnexpectedResponses != 0) {
		t.Errorf("full capacity rejection mismatch expected=%d total=%d unexpected=%d invalid_json=%d", fullExpectedRejections, len(gatewayTimes), fullUnexpectedResponses, fullInvalidJSON)
	}
	if initialActive && scenario == "normal" && upstreamRequests.Load() > upstreamBefore && completed-completedBefore-cleanupRemaining == 0 {
		t.Error("active protection produced no completed facts")
	}
	if adminErrors > 0 || (scenario != "full" && gatewayErrors > 0) || fullUnexpectedResponses > 0 {
		t.Errorf("HTTP failures gateway=%d admin=%d", gatewayErrors, adminErrors)
	}
}
